import asyncio
import subprocess
import yaml
from pathlib import Path
from config import PipelineConfig, load_env


class Pipeline:
    """Orchestrates CI/CD stages: build, test, deploy."""

    def __init__(self, config_path: str = "pipeline.yml"):
        with open(config_path) as f:
            self.stages = yaml.safe_load(f)["stages"]
        self.config = PipelineConfig.from_env()

    async def run_stage(self, name: str) -> int:
        # Execute each command in the stage
        cmds = self.stages[name]["commands"]
        for cmd in cmds:
            proc = await asyncio.create_subprocess_shell(
                cmd, stdout=subprocess.PIPE, stderr=subprocess.PIPE
            )
            stdout, _ = await proc.communicate()
            print(f"[{name}] {stdout.decode().strip()}")
        return 0
